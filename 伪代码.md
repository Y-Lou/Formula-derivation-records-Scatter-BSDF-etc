// 立体角计算
double SolidAngle(double iRed1,double iRed2)
{
    // iRed1 < iRed2
    return cos(iRed1) - cos(iRed2);
}
// 角度差计算
double T(int iIndex, double iStep, double iRed)
{
    double theta1 = iIndex*iStep;
    double theta2 = (iIndex+1)*iStep;
    return (iRed - theta1) / (theta2 - theta1);
}
// 球面线性插值计算Row
double CircleLineRow(double iRed, double iOriginStep, double iNewStep, int iCol)
{
    int originIndex1 = std::flood(iRed / iOriginStep);
    int originIndex2 = originIndex1+1;
    if (originIndex1 == 0 || originIndex1 == this->Rows-1) return this->pMatrix->at(originIndex1, iCol);
    double I1 = this->pMatrix->at(originIndex1, iCol);
    double I2 = this->pMatrix->at(originIndex2, iCol);
    double t = T(originIndex1, iOriginStep, iRed);
    double F1 = I1 * SKM::Mathf::Sin((1-t)*iRed) / SKM::Mathf::Sin(iRed);
    double F2 = I2 * SKM::Mathf::Sin(t*iRed) / SKM::Mathf::Sin(iRed);
    return F1+F2;
}
// 球面线性插值计算Col
double CircleLineCol(double iRed, double iOriginStep, double iNewStep, int iRow)
{
    int originIndex1 = std::flood(iRed / iOriginStep);
    if (originIndex1 == 0 || originIndex1 == this->Cols-1) return this->pMatrix->at(iRow, originIndex1);
    int originIndex2 = originIndex1+1;
    double I1 = this->pMatrix->at(iRow, originIndex1);
    double I2 = this->pMatrix->at(iRow, originIndex2);
    double t = T(originIndex1, iOriginStep, iRed);
    double F1 = I1 * SKM::Mathf::Sin((1-t)*iRed) / SKM::Mathf::Sin(iRed);
    double F2 = I2 * SKM::Mathf::Sin(t*iRed) / SKM::Mathf::Sin(iRed);
    return F1+F2;
}
// 球面线性插值计算
void CircleLine(SKM::Matrixf& iMatrix, double iOriginStep, double iNewStep)
{
    // 先对每行数据进行插值
    SKM::Matrixf matrixRow;
    matrixRow.SetSize(this->Rows, iMatrix.GetCols());
    double iRed = 0.0;
    for (int i = 0; i < this->Rows; i++)
    {
        for (int j = 0; j < iMatrix.GetCols(); j++)
        {
            iRed = j * iNewStep;
            matrixRow.at(i,j) = CircleLineCol(iRed, iOriginStep, iNewStep, i);
        }
    }
    // 在对每一列数据进行插值
    for (int i = 0; i < iMatrix.GetCols(); i++)
    {
        for (int j = 0; j < iMatrix.GetRows(); j++)
        {
            iRed = j * iNewStep;
            iMatrix.at(i,j) = matrixRow.CircleLineRow(iRed, iOriginStep, iNewStep, i);
        }
    }
}

double CircleSolidAngleRow(double iRed, double iOriginStep, double iNewStep, int iCol)
{
    // 计算原始数据的观测角度
	int originIndex1 = std::flood(iRed / iOriginStep);
    if (originIndex1 == 0 || originIndex1 == this->Rows-1) return this->pMatrix->at(originIndex1, iCol);
    int originIndex2 = originIndex1+1;
    // 角度计算
    double red1 = originIndex1 * iOriginStep;
    double red2 = originIndex2 * iOriginStep;
    // 立体角计算
    double solidAngle1 = SolidAngle(red1, iRed);
    double soildAngle2 = SolidAngle(iRed, red2);
    // 光强计算
    double I1 = this->pMatrix->at(originIndex1, iCol);
    double I2 = this->pMatrix->at(originIndex2, iCol);
    // 插值计算
    double soildAngle = solidAngle1 + soildAngle2;
    return (solidAngle1 * I1 + soildAngle2 * I2) / soildAngle;
}
